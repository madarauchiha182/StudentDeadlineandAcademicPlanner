System, Data & Workflow Architecture Models

Formal system engineering models representing the Student Deadline & Academic Planner designed at JSPM's RSCOE Pune.

1. Domain Entity & Class Model (UML)

class User { +String id +String rollNo +String name +String email +String passwordHash +UserRole role [STUDENT, FACULTY, ADMIN] +String batch +login(credentials): SessionToken } class Student extends User { +addTask(taskData): Task +updateTaskStatus(taskId, status): void +snoozeAlert(alertId, duration): void +viewCalendar(viewType): CalendarView } class Faculty extends User { +postClassDeadline(batchId, taskData): Task +generateClassReport(batchId, subjectId): Report } class Task { +String id +String title +String subjectCode +TaskType type [ASSIGNMENT, EXAM, PROJECT, ACTIVITY] +Date dueDate +Time dueTime +PriorityLevel priority [HIGH, MEDIUM, LOW] +TaskStatus status [NOT_STARTED, IN_PROGRESS, COMPLETED, OVERDUE] +Boolean isFacultyPosted +calculateProximity(): Integer } class Alert { +String id +String taskId +AlertType type [3_DAYS, 1_DAY, SAME_DAY, OVERDUE] +Boolean isSnoozed +DateTime snoozedUntil +snooze(hours): void +dismiss(): void }

2. PostgreSQL Relational Database Schema Model (DDL)

CREATE TABLE users ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), roll_no VARCHAR(50) UNIQUE NOT NULL, full_name VARCHAR(120) NOT NULL, email VARCHAR(120) UNIQUE NOT NULL, password_hash VARCHAR(255) NOT NULL, role VARCHAR(20) NOT NULL CHECK (role IN ('student', 'faculty', 'admin')), batch_id VARCHAR(20), created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE tasks ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), student_id UUID REFERENCES users(id) ON DELETE CASCADE, title VARCHAR(200) NOT NULL, subject_code VARCHAR(30) NOT NULL, task_type VARCHAR(40) NOT NULL, due_date DATE NOT NULL, due_time TIME DEFAULT '23:59:00', priority VARCHAR(10) NOT NULL CHECK (priority IN ('High', 'Medium', 'Low')), status VARCHAR(20) NOT NULL CHECK (status IN ('Not started', 'In progress', 'Completed', 'Overdue')), is_faculty_posted BOOLEAN DEFAULT FALSE, description TEXT, created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE alerts ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), task_id UUID REFERENCES tasks(id) ON DELETE CASCADE, alert_title VARCHAR(200) NOT NULL, alert_type VARCHAR(30) NOT NULL, is_snoozed BOOLEAN DEFAULT FALSE, snoozed_until TIMESTAMP WITH TIME ZONE, is_dismissed BOOLEAN DEFAULT FALSE, created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP ); CREATE INDEX idx_tasks_due_date ON tasks(due_date); CREATE INDEX idx_tasks_student ON tasks(student_id);
